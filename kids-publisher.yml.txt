name: Kids Story Publisher

on:
  schedule:
    - cron: "0 */6 * * *"   # كل 6 ساعات (UTC). غيّرها من هنا
  workflow_dispatch:
    inputs:
      dry_run:
        description: "ابني الفيديو بدون رفع"
        type: boolean
        default: false
      privacy:
        description: "public / unlisted / private"
        default: "public"
      make_short:
        description: "اعمل Short"
        type: boolean
        default: true

permissions:
  contents: write

concurrency:
  group: kids-publisher
  cancel-in-progress: false

jobs:
  publish:
    runs-on: ubuntu-latest
    timeout-minutes: 45
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: pip
      - name: Install system deps
        run: sudo apt-get update && sudo apt-get install -y ffmpeg
      - name: Install python deps
        run: pip install -r requirements.txt
      - name: Run bot
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
          GEMINI_MODEL: ${{ vars.GEMINI_MODEL || 'gemini-2.5-flash,gemini-2.0-flash' }}
          PEXELS_API_KEY: ${{ secrets.PEXELS_API_KEY }}
          YT_TOKEN_JSON: ${{ secrets.YT_TOKEN_JSON }}
          TTS_VOICE: ${{ vars.TTS_VOICE || 'ar-EG-SalmaNeural' }}
          TTS_RATE: ${{ vars.TTS_RATE || '-10%' }}
          PRIVACY_STATUS: ${{ github.event.inputs.privacy || vars.PRIVACY_STATUS || 'public' }}
          MAKE_SHORT: ${{ github.event.inputs.make_short || 'true' }}
          DRY_RUN: ${{ github.event.inputs.dry_run || 'false' }}
        run: python kids_youtube_uploader.py
      - name: Save history
        if: success()
        run: |
          git config user.name "kids-bot"
          git config user.email "kids-bot@users.noreply.github.com"
          git add kids_history.json || true
          git diff --cached --quiet || (git commit -m "update history" && git pull --rebase && git push)
      - name: Upload artifacts
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: output
          path: out/*.mp4
          retention-days: 3
          if-no-files-found: ignore
